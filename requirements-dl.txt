# pytorch-forecasting 1.8 caps pandas below 3.1. requirements.txt allows

# environment that trains (and backtests) the TFT, not on the trader
# unless it will run ml_tft.
#
# torch from PyPI is the CUDA build (a few GB with its nvidia-* wheels); on
# a CPU-only machine, install it first from the CPU index instead:
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
